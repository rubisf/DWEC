/**
 * @typedef {Object} Estadisticas
 * @property {number} media - Media aritmética de los valores.
 * @property {number} maximo - Valor más alto de la serie.
 * @property {number} minimo - Valor más bajo de la serie.
 */

/**
 * Calcula métricas estadísticas descriptivas sobre una colección numérica.
 * @param {number[]} numeros - Array con valores numéricos.
 * @returns {Estadisticas} Objeto con las métricas calculadas.
 * @throws {Error} Si el array está vacío o contiene elementos no numéricos.
 * @example
 * const stats = calcularEstadisticas([10, 20, 30]); // { media: 20, maximo: 30, minimo: 10 }
 */
function calcularEstadisticas(numeros) {
    if (!Array.isArray(numeros) || numeros.length === 0) {
        throw new Error("El argumento debe ser un array con al menos un elemento numérico.");
    }

    console.time("Cálculo Estadístico");

    let suma = 0;
    for (const num of numeros) {
        if (typeof num !== "number" || Number.isNaN(num)) {
            throw new TypeError("Todos los elementos del array deben ser números válidos.");
        }
        suma += num;
    }

    const media = Number((suma / numeros.length).toFixed(2));
    const maximo = Math.max(...numeros);
    const minimo = Math.min(...numeros);

    // Punto de interrupción para inspección en navegador
    debugger;

    console.timeEnd("Cálculo Estadístico");

    return { media, maximo, minimo };
}

// Pruebas y Aserciones
const muestra = [15, 22, 8, 45, 31, 19];
const resultado = calcularEstadisticas(muestra);

console.assert(resultado.maximo === 45, "Error en cálculo de máximo");
console.assert(resultado.minimo === 8, "Error en cálculo de mínimo");
console.table(resultado);
