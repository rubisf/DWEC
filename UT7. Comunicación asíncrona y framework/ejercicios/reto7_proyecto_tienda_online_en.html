Reto 7: Proyecto Tienda Online en Angular: Menú, Catálogo con Tarjetas y Vista de Detalle

📝 Enunciado:

Diseña un simulador interactivo completo que represente la aplicación Tienda Online en Angular, incluyendo: 1) Barra de navegación superior con menú, 2) Vista de catálogo que renderice tarjetas de productos con imagen, título, precio y botón de navegación, y 3) Vista de detalle de producto que simule el cambio de ruta (Angular Router con `:id`) mostrando la ficha ampliada del producto con opción de retorno al catálogo.

💡 Pistas: Crea un estado visual de vista ("catalogo" | "detalle") y almacena el productoSeleccionado. Al hacer clic en una tarjeta, cambia el estado a "detalle" mostrando la vista ampliada.
Salida dual: Terminal interactivo y Consola F12
Terminal de Ejecución en Vivo
0 entradas
🖥️ Área de Renderizado DOM / Contenedor de Pruebas #sandbox-container
📖 Explicación Técnica y Código Fuente Ver código

El desacoplamiento entre componentes modulares, el enrutador SPA y los servicios inyectables constituye la arquitectura estándar para aplicaciones comerciales desarrolladas en Angular.

// Reto 7: Simulador Interactivo de la Aplicación Tienda Online en Angular (Menú, Catálogo y Vista de Detalle)
class TiendaAngularApp {
    constructor(contenedorId) {
        this.contenedor = document.getElementById(contenedorId);
        this.vistaActual = "catalogo"; // 'catalogo' | 'detalle'
        this.productoSeleccionadoId = null;

        this.productos = [
            { id: 1, nombre: "Portátil Angular Ultra", precio: 1249, imagen: "💻", descripcion: "Procesador de última generación 16-Core, 32GB RAM DDR5, Pantalla OLED 120Hz. Ideal para desarrollo frontend intensivo con Angular CLI.", categoria: "Ordenadores" },
            { id: 2, nombre: "Teclado Mecánico TypeScript", precio: 129, imagen: "⌨️", descripcion: "Switches táctiles silenciosos, iluminación RGB direccionable por software, chasis de aluminio pulido y cable trenzado USB-C.", categoria: "Periféricos" },
            { id: 3, nombre: "Monitor Curvo 34\" RxJS", precio: 499, imagen: "🖥️", descripcion: "Resolución UWQHD 3440x1440, tasa de refresco 144Hz, cobertura 99% sRGB. Visualización multitarea fluida sin parpadear.", categoria: "Pantallas" },
            { id: 4, nombre: "Auriculares Studio Signals", precio: 189, imagen: "🎧", descripcion: "Audio de alta resolución Hi-Res, cancelación activa de ruido adaptativa, micrófonos beamforming para videollamadas claras.", categoria: "Audio" }
        ];

        this.inyectarEstilos();
    }

    inyectarEstilos() {
        if (document.getElementById("tienda-angular-styles")) return;
        const style = document.createElement("style");
        style.id = "tienda-angular-styles";
        style.textContent = `
            .t-app { font-family: system-ui, sans-serif; border-radius: 10px; overflow: hidden; background: #1e293b; color: #f8fafc; text-align: left; }
            .t-nav { background: #0f172a; border-bottom: 1px solid #334155; padding: 12px 20px; display: flex; justify-content: space-between; align-items: center; }
            .t-nav-brand { display: flex; align-items: center; gap: 10px; font-weight: bold; color: #f8fafc; }
            .t-nav-btn { background: #2563eb; color: white; border: none; padding: 6px 14px; border-radius: 6px; cursor: pointer; font-weight: bold; font-size: 0.85em; }
            .t-nav-btn.inactive { background: transparent; }
            .t-body { padding: 20px; }
            .t-grid { display: grid; grid-template-columns: repeat(auto-fill, minmax(200px, 1fr)); gap: 14px; margin-top: 14px; }
            .t-card { background: #0f172a; border: 1px solid #334155; border-radius: 10px; padding: 14px; display: flex; flex-direction: column; justify-content: space-between; }
            .t-badge { font-size: 0.7em; background: #0284c7; color: white; padding: 2px 6px; border-radius: 4px; font-weight: bold; display: inline-block; }
            .t-price { color: #38bdf8; font-weight: bold; font-size: 1.1em; margin: 4px 0 0 0; }
            .t-btn-detail { margin-top: 14px; background: #2563eb; color: white; border: none; padding: 8px; border-radius: 6px; cursor: pointer; font-weight: bold; font-size: 0.85em; width: 100%; }
            .t-btn-back { background: #334155; color: white; border: none; padding: 6px 12px; border-radius: 6px; cursor: pointer; margin-bottom: 16px; font-size: 0.85em; font-weight: bold; }
            .t-detail-card { background: #0f172a; border: 1px solid #38bdf8; border-radius: 12px; padding: 20px; display: grid; grid-template-columns: 80px 1fr; gap: 20px; align-items: start; }
            .t-detail-desc { color: #cbd5e1; font-size: 0.9em; line-height: 1.5; background: #1e293b; padding: 12px; border-radius: 8px; border-left: 3px solid #38bdf8; margin: 12px 0; }
            .t-btn-cart { background: #10b981; color: white; border: none; padding: 10px 18px; border-radius: 6px; cursor: pointer; font-weight: bold; font-size: 0.9em; }
        `;
        document.head.appendChild(style);
    }

    render() {
        if (!this.contenedor) return;
        let contenidoHtml = "";

        const navbarHtml = `
            <nav class="t-nav">
                <div class="t-nav-brand">
                    <span style="font-size: 1.5em;">🅰️</span>
                    <span>Tienda Angular Store</span>
                </div>
                <div>
                    <button id="nav-catalogo" class="t-nav-btn ${this.vistaActual === 'catalogo' ? '' : 'inactive'}">
                        🛒 Catálogo
                    </button>
                </div>
            </nav>
        `;

        if (this.vistaActual === "catalogo") {
            contenidoHtml = `
                <div class="t-body">
                    <div style="margin-bottom: 16px;">
                        <h2 style="margin: 0; color: #f8fafc; font-size: 1.2em;">Catálogo de Productos Destacados</h2>
                        <p style="margin: 4px 0 0 0; color: #94a3b8; font-size: 0.85em;">Selecciona un producto para inspeccionar sus especificaciones técnicas (Angular Router: /producto/:id)</p>
                    </div>
                    <div class="t-grid">
                        ${this.productos.map(p => `
                            <div class="t-card">
                                <div>
                                    <div style="font-size: 3em; text-align: center; margin-bottom: 10px;">${p.imagen}</div>
                                    <span class="t-badge">${p.categoria}</span>
                                    <h3 style="margin: 8px 0 4px 0; color: #f8fafc; font-size: 1em;">${p.nombre}</h3>
                                    <p class="t-price">${p.precio} €</p>
                                </div>
                                <button class="btn-ver-detalle t-btn-detail" data-id="${p.id}">
                                    🔍 Ver Detalle
                                </button>
                            </div>
                        `).join('')}
                    </div>
                </div>
            `;
        } else if (this.vistaActual === "detalle") {
            const producto = this.productos.find(p => p.id === this.productoSeleccionadoId);
            if (producto) {
                contenidoHtml = `
                    <div class="t-body">
                        <button id="btn-volver" class="t-btn-back">
                            ← Volver al Catálogo
                        </button>
                        <div class="t-detail-card">
                            <div style="font-size: 3.5em; text-align: center; background: #1e293b; border-radius: 10px; padding: 10px;">${producto.imagen}</div>
                            <div>
                                <div style="display: flex; gap: 8px; align-items: center; margin-bottom: 6px;">
                                    <span class="t-badge">${producto.categoria}</span>
                                    <span style="font-size: 0.75em; color: #94a3b8;">Ruta activa: /producto/${producto.id}</span>
                                </div>
                                <h2 style="margin: 0 0 8px 0; color: #f8fafc; font-size: 1.4em;">${producto.nombre}</h2>
                                <p class="t-price" style="font-size: 1.3em;">${producto.precio} €</p>
                                <p class="t-detail-desc">
                                    ${producto.descripcion}
                                </p>
                                <div style="margin-top: 16px; display: flex; gap: 10px;">
                                    <button onclick="alert('🛒 Producto añadido al carrito de compra de Angular')" class="t-btn-cart">
                                        🛒 Añadir al Carrito
                                    </button>
                                </div>
                            </div>
                        </div>
                    </div>
                `;
            }
        }

        this.contenedor.className = "t-app";
        this.contenedor.innerHTML = navbarHtml + contenidoHtml;
        this.bindEvents();
    }

    bindEvents() {
        const btnNavCat = this.contenedor.querySelector("#nav-catalogo");
        if (btnNavCat) {
            btnNavCat.onclick = () => {
                this.vistaActual = "catalogo";
                this.render();
            };
        }

        const btnVolver = this.contenedor.querySelector("#btn-volver");
        if (btnVolver) {
            btnVolver.onclick = () => {
                this.vistaActual = "catalogo";
                this.render();
            };
        }

        const btnsDetalle = this.contenedor.querySelectorAll(".btn-ver-detalle");
        btnsDetalle.forEach(btn => {
            btn.onclick = (e) => {
                const id = Number(e.currentTarget.dataset.id);
                this.productoSeleccionadoId = id;
                this.vistaActual = "detalle";
                this.render();
            };
        });
    }
}

// Inicialización de la Tienda Angular en el Sandbox
const mountNode = document.getElementById("app") || document.getElementById("sandbox-container");
if (mountNode) {
    const tienda = new TiendaAngularApp(mountNode.id);
    tienda.render();
    console.log("🅰️ [Angular Store App] Tienda Online montada con éxito con Menú, Catálogo de Tarjetas y Vista de Detalle.");
}
