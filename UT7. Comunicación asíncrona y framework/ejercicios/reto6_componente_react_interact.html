Reto 6: Componente React Interactivo de Lista de Tareas y Filtro con useState e Inmutabilidad

📝 Enunciado:

Diseña un componente funcional en React llamado `GestorTareasReact` que implemente: 1) Un estado para almacenar un array de tareas (con id, texto, completada y prioridad: "alta" | "media" | "baja"), 2) Un estado para el filtro activo ("todas", "pendientes", "completadas"), 3) Funcionalidad para alternar el estado completada de una tarea de forma estrictamente inmutable con spread operator, 4) Añadir nuevas tareas desde un formulario controlado, 5) Eliminar tareas seleccionadas, 6) Un resumen dinámico con contador de pendientes y porcentaje de avance, y 7) Renderizado del componente montándolo en `#app` con `ReactDOM.createRoot`.

💡 Pistas: Utiliza React.useState([]) para las tareas y React.useState("todas") para el filtro. En el toggle, usa tareas.map(t => t.id === id ? { ...t, completada: !t.completada } : t). Para añadir, usa setTareas(prev => [...prev, nuevaTarea]). Monta con ReactDOM.createRoot(document.getElementById("app")).render(React.createElement(GestorTareasReact)).

La inmutabilidad en la actualización del estado de React es el pilar que permite al Virtual DOM detectar mutaciones por referencia y reconciliar únicamente los nodos afectados, garantizando un rendimiento óptimo.

// Reto 6: Componente React Funcional con Gestión de Estado (useState) e Inmutabilidad
function GestorTareasReact() {
    const [tareas, setTareas] = React.useState([
        { id: 1, texto: "Comprender la reconciliación del Virtual DOM", completada: true, prioridad: "alta" },
        { id: 2, texto: "Gestionar estado inmutable con useState", completada: true, prioridad: "alta" },
        { id: 3, texto: "Consumir APIs asíncronas dentro de useEffect", completada: false, prioridad: "media" },
        { id: 4, texto: "Evitar fugas de memoria con AbortController", completada: false, prioridad: "baja" }
    ]);
    const [filtro, setFiltro] = React.useState("todas");
    const [nuevoTexto, setNuevoTexto] = React.useState("");
    const [prioridad, setPrioridad] = React.useState("media");

    // Operación Inmutable: Alternar completado con .map() y operador spread
    const alternarTarea = (id) => {
        setTareas(prev => prev.map(t => t.id === id ? { ...t, completada: !t.completada } : t));
    };

    // Operación Inmutable: Eliminar tarea con .filter()
    const eliminarTarea = (id) => {
        setTareas(prev => prev.filter(t => t.id !== id));
    };

    // Operación Inmutable: Añadir tarea con [...prev, nueva]
    const agregarTarea = (e) => {
        e.preventDefault();
        const textoLimpio = nuevoTexto.trim();
        if (!textoLimpio) return;

        const nueva = {
            id: Date.now(),
            texto: textoLimpio,
            completada: false,
            prioridad
        };

        setTareas(prev => [...prev, nueva]);
        setNuevoTexto("");
    };

    // Filtrado declarativo en render
    const tareasVisibles = tareas.filter(t => {
        if (filtro === "pendientes") return !t.completada;
        if (filtro === "completadas") return t.completada;
        return true;
    });

    const totalPendientes = tareas.filter(t => !t.completada).length;
    const porcentajeProgreso = tareas.length > 0
        ? Math.round(((tareas.length - totalPendientes) / tareas.length) * 100)
        : 0;

    return React.createElement("div", {
        className: "p-4 bg-slate-900 border border-slate-700 rounded-xl text-slate-100 font-sans max-w-xl mx-auto shadow-2xl"
    },
        // Encabezado con Métricas
        React.createElement("div", { className: "flex items-center justify-between border-b border-slate-700 pb-3 mb-3" },
            React.createElement("div", null,
                React.createElement("h3", { className: "text-sm font-bold text-indigo-400 flex items-center gap-2 m-0" }, "⚛️ Gestor de Tareas React"),
                React.createElement("span", { className: "text-[11px] text-slate-400" }, "Estado local reactivo con Hook useState")
            ),
            React.createElement("span", { className: "text-xs bg-indigo-950 text-indigo-300 border border-indigo-700 px-2.5 py-1 rounded-full font-semibold" },
                `${totalPendientes} pendientes (${porcentajeProgreso}%)`
            )
        ),

        // Formulario controlado para añadir tarea
        React.createElement("form", { onSubmit: agregarTarea, className: "flex gap-2 mb-3" },
            React.createElement("input", {
                type: "text",
                placeholder: "Escribe una nueva tarea...",
                value: nuevoTexto,
                onChange: (e) => setNuevoTexto(e.target.value),
                className: "flex-1 bg-slate-950 border border-slate-700 rounded-lg px-3 py-1.5 text-xs text-white placeholder-slate-500 focus:outline-none focus:border-indigo-500"
            }),
            React.createElement("select", {
                value: prioridad,
                onChange: (e) => setPrioridad(e.target.value),
                className: "bg-slate-950 border border-slate-700 rounded-lg px-2 py-1.5 text-xs text-slate-300"
            },
                React.createElement("option", { value: "baja" }, "Baja"),
                React.createElement("option", { value: "media" }, "Media"),
                React.createElement("option", { value: "alta" }, "Alta")
            ),
            React.createElement("button", {
                type: "submit",
                className: "bg-indigo-600 hover:bg-indigo-500 text-white font-bold px-3 py-1.5 rounded-lg text-xs transition-colors"
            }, "+ Añadir")
        ),

        // Barra de Filtros
        React.createElement("div", { className: "flex gap-1.5 mb-3" },
            ["todas", "pendientes", "completadas"].map(f =>
                React.createElement("button", {
                    key: f,
                    onClick: () => setFiltro(f),
                    className: `text-xs px-2.5 py-1 rounded-md font-semibold capitalize transition-all ${
                        filtro === f ? "bg-indigo-600 text-white shadow-sm" : "bg-slate-800 text-slate-400 hover:bg-slate-700"
                    }`
                }, f)
            )
        ),

        // Lista Reactiva con keys
        React.createElement("ul", { className: "space-y-1.5 max-h-52 overflow-y-auto p-0 m-0 list-none" },
            tareasVisibles.length === 0
                ? React.createElement("li", { className: "text-slate-500 text-xs italic py-3 text-center" }, "No hay tareas en esta vista.")
                : tareasVisibles.map(t =>
                    React.createElement("li", {
                        key: t.id,
                        className: `flex items-center justify-between p-2 rounded-lg border text-xs transition-all ${
                            t.completada ? "bg-slate-950/40 border-slate-800 text-slate-500 line-through" : "bg-slate-800/80 border-slate-700 text-slate-200"
                        }`
                    },
                        React.createElement("span", {
                            onClick: () => alternarTarea(t.id),
                            className: "cursor-pointer flex-1 select-none flex items-center gap-1.5"
                        },
                            t.completada ? "✅" : "⭕",
                            t.texto
                        ),
                        React.createElement("div", { className: "flex items-center gap-2" },
                            React.createElement("span", {
                                className: `text-[10px] px-1.5 py-0.5 rounded font-bold uppercase ${
                                    t.prioridad === "alta" ? "bg-red-950 text-red-300 border border-red-800" :
                                    t.prioridad === "media" ? "bg-amber-950 text-amber-300 border border-amber-800" :
                                    "bg-slate-800 text-slate-400"
                                }`
                            }, t.prioridad),
                            React.createElement("button", {
                                onClick: () => eliminarTarea(t.id),
                                className: "text-slate-400 hover:text-red-400 text-xs px-1 bg-transparent border-0 cursor-pointer",
                                title: "Eliminar tarea"
                            }, "🗑️")
                        )
                    )
                )
        )
    );
}

// Montaje en el contenedor DOM (#app)
const target = document.getElementById("app") || document.getElementById("sandbox-container");
if (target) {
    const root = ReactDOM.createRoot(target);
    root.render(React.createElement(GestorTareasReact));
    console.log("⚛️ [React 18] GestorTareasReact montado con éxito en el DOM.");
}
