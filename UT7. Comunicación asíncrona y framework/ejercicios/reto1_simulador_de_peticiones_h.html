Reto 1: Simulador de Peticiones HTTP y Clasificador de Respuestas por Código de Estado

📝 Enunciado:

Escribe una función llamada `procesarRespuestaServidor(respuestaHttp)` que reciba un objeto simulando una respuesta HTTP (con propiedades `status`, `statusText`, `headers` y `data`). Debe clasificar la respuesta en una de las 4 familias (Éxito 2xx, Redirección 3xx, Error de Cliente 4xx, Error de Servidor 5xx), extraer el tipo de contenido desde la cabecera `Content-Type`, y devolver un informe estructurado listo para mostrar al usuario.

💡 Pistas: Usa Math.floor(respuesta.status / 100) para obtener el dígito principal de la familia (2, 3, 4 o 5). Valida el statusText y devuelve un objeto con { familia, esValido, mensaje }.

La correcta interpretación de los códigos de estado HTTP es indispensable para que el frontend gestione la navegación y notifique errores al usuario de forma contextualizada.

function procesarRespuestaServidor(respuesta) {
    const status = respuesta.status;
    const familiaCodigo = Math.floor(status / 100);
    const contentType = respuesta.headers?.["Content-Type"] || "desconocido";

    let diagnostico = {
        status,
        statusText: respuesta.statusText || "Sin descripción",
        contentType,
        esExito: false,
        categoria: "",
        mensajeUsuario: ""
    };

    switch (familiaCodigo) {
        case 2:
            diagnostico.esExito = true;
            diagnostico.categoria = "2xx - Éxito";
            diagnostico.mensajeUsuario = status === 201 
                ? "Registro creado correctamente." 
                : "Operación realizada con éxito.";
            break;
        case 3:
            diagnostico.categoria = "3xx - Redirección";
            diagnostico.mensajeUsuario = "El recurso ha sido reubicado.";
            break;
        case 4:
            diagnostico.categoria = "4xx - Error de Cliente";
            diagnostico.mensajeUsuario = status === 404 
                ? "El elemento solicitado no existe." 
                : "Solicitud incorrecta o sin autorización suficiente.";
            break;
        case 5:
            diagnostico.categoria = "5xx - Error de Servidor";
            diagnostico.mensajeUsuario = "Fallo interno en el servidor. Reintentando en unos momentos...";
            break;
        default:
            diagnostico.categoria = "Código Desconocido";
            diagnostico.mensajeUsuario = "Respuesta de red no estandarizada.";
    }

    console.log("=== Diagnóstico de Respuesta HTTP ===");
    console.table(diagnostico);
    return diagnostico;
}
