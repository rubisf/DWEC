Reto 7: Componente React Asíncrono de Catálogo de Usuarios con useEffect, Búsqueda en Vivo y Cancelación con AbortController

📝 Enunciado:

Diseña un componente funcional en React llamado `CatalogoUsuariosAPI` que implemente: 1) Consumo de la API pública `https://jsonplaceholder.typicode.com/users` dentro de un hook `useEffect` con `fetch`, 2) Gestión de tres estados reactivos (`usuarios`, `cargando`, `error`), 3) Una barra de búsqueda en tiempo real que filtre a los usuarios por nombre sin requerir nuevas peticiones de red, 4) Función de limpieza con `AbortController` para cancelar la petición saliente si el componente se desmonta, 5) Botón interactivo de recarga manual de datos, y 6) Renderizado de tarjetas de usuario con nombre, empresa, email y ciudad, montando la aplicación en `#app`.

💡 Pistas: Instancia const controller = new AbortController() al inicio de useEffect y pasa { signal: controller.signal } en fetch. En el bloque catch, comprueba if (err.name === "AbortError") return. Retorna () => controller.abort() como función de limpieza. Renderiza condicionalmente el spinner, la alerta de error o la cuadrícula de usuarios filtrados.

El desacoplamiento de la obtención de datos mediante useEffect y la cancelación de peticiones obsoletas con AbortController garantiza aplicaciones web reactivas robustas, sin condiciones de carrera ni fugas de memoria.

// Reto 7: Componente React Asíncrono con useEffect, API Fetch y AbortController
function CatalogoUsuariosAPI() {
    const [usuarios, setUsuarios] = React.useState([]);
    const [cargando, setCargando] = React.useState(true);
    const [error, setError] = React.useState(null);
    const [busqueda, setBusqueda] = React.useState("");
    const [recargaTick, setRecargaTick] = React.useState(0);

    // Efecto asíncrono para consumir la API REST con ciclo de vida controlado
    React.useEffect(() => {
        const controller = new AbortController();

        async function fetchUsuarios() {
            setCargando(true);
            setError(null);
            console.log("🌐 [React useEffect] Iniciando petición HTTP a JSONPlaceholder...");

            try {
                const res = await fetch("https://jsonplaceholder.typicode.com/users", {
                    signal: controller.signal,
                    headers: { "Accept": "application/json" }
                });

                if (!res.ok) {
                    throw new Error(`Respuesta HTTP no satisfactoria (${res.status} ${res.statusText})`);
                }

                const data = await res.json();
                setUsuarios(data);
                console.log(`✅ [React useEffect] Recibidos ${data.length} usuarios con éxito.`);
            } catch (err) {
                if (err.name === "AbortError") {
                    console.log("🛑 [React useEffect] Petición Fetch abortada limpiamente por desmontaje.");
                } else {
                    console.error("❌ [React useEffect] Error en la petición:", err.message);
                    setError(err.message);
                }
            } finally {
                setCargando(false);
            }
        }

        fetchUsuarios();

        // Función de Limpieza (Cleanup Function)
        return () => {
            controller.abort();
        };
    }, [recargaTick]); // Se re-ejecuta cuando el usuario solicita una recarga

    // Filtrado en memoria reactivo
    const usuariosFiltrados = usuarios.filter(u =>
        u.name.toLowerCase().includes(busqueda.toLowerCase()) ||
        u.email.toLowerCase().includes(busqueda.toLowerCase())
    );

    return React.createElement("div", {
        className: "p-4 bg-slate-900 border border-slate-700 rounded-xl text-slate-100 font-sans max-w-2xl mx-auto shadow-2xl"
    },
        // Encabezado
        React.createElement("div", { className: "flex items-center justify-between border-b border-slate-700 pb-3 mb-4 flex-wrap gap-2" },
            React.createElement("div", null,
                React.createElement("h3", { className: "text-sm font-bold text-sky-400 flex items-center gap-2 m-0" },
                    "👥 Catálogo de Usuarios Asíncrono"
                ),
                React.createElement("span", { className: "text-[11px] text-slate-400" },
                    "React 18 · useEffect · Fetch API · AbortController"
                )
            ),
            React.createElement("button", {
                onClick: () => setRecargaTick(t => t + 1),
                disabled: cargando,
                className: "bg-sky-600 hover:bg-sky-500 disabled:bg-slate-700 text-white text-xs font-semibold px-3 py-1.5 rounded-lg transition-colors cursor-pointer border-0 flex items-center gap-1.5"
            },
                cargando ? "⏳ Cargando..." : "🔄 Recargar API"
            )
        ),

        // Barra de búsqueda interactiva
        React.createElement("div", { className: "mb-4" },
            React.createElement("input", {
                type: "text",
                placeholder: "🔍 Filtrar por nombre o email en tiempo real...",
                value: busqueda,
                onChange: (e) => setBusqueda(e.target.value),
                className: "w-full bg-slate-950 border border-slate-700 rounded-lg px-3 py-2 text-xs text-white placeholder-slate-500 focus:outline-none focus:border-sky-500 box-border"
            })
        ),

        // Estado 1: Cargando (Skeleton / Spinner)
        cargando && React.createElement("div", { className: "p-6 text-center text-slate-400 space-y-2" },
            React.createElement("div", { className: "text-2xl animate-spin inline-block" }, "⏳"),
            React.createElement("p", { className: "text-xs m-0" }, "Consultando API REST externa y sincronizando Virtual DOM...")
        ),

        // Estado 2: Error
        error && !cargando && React.createElement("div", {
            className: "p-4 bg-red-950/50 border border-red-800 rounded-lg text-red-200 text-xs"
        },
            React.createElement("strong", null, "❌ Error al cargar usuarios: "),
            error,
            React.createElement("div", { className: "mt-2" },
                React.createElement("button", {
                    onClick: () => setRecargaTick(t => t + 1),
                    className: "bg-red-700 hover:bg-red-600 text-white text-xs px-3 py-1 rounded border-0 cursor-pointer"
                }, "Reintentar ahora")
            )
        ),

        // Estado 3: Éxito con Cuadrícula de Usuarios
        !cargando && !error && React.createElement("div", null,
            React.createElement("div", { className: "text-[11px] text-slate-400 mb-2 flex justify-between" },
                React.createElement("span", null, `Mostrando ${usuariosFiltrados.length} de ${usuarios.length} usuarios`),
                busqueda && React.createElement("span", { className: "text-sky-400" }, `Filtro: "${busqueda}"`)
            ),
            usuariosFiltrados.length === 0
                ? React.createElement("p", { className: "text-slate-500 text-xs italic text-center py-4" }, "No se encontraron coincidencias con la búsqueda.")
                : React.createElement("div", { className: "grid grid-cols-1 sm:grid-cols-2 gap-2.5 max-h-64 overflow-y-auto pr-1" },
                    usuariosFiltrados.map(u =>
                        React.createElement("div", {
                            key: u.id,
                            className: "p-3 bg-slate-800/80 border border-slate-700/80 rounded-lg text-xs hover:border-sky-500/50 transition-all flex flex-col justify-between"
                        },
                            React.createElement("div", null,
                                React.createElement("div", { className: "font-bold text-slate-100 flex items-center justify-between mb-1" },
                                    React.createElement("span", null, `👤 ${u.name}`),
                                    React.createElement("span", { className: "text-[10px] text-slate-400 font-mono" }, `#${u.id}`)
                                ),
                                React.createElement("p", { className: "text-slate-400 text-[11px] m-0" }, `📧 ${u.email}`),
                                React.createElement("p", { className: "text-slate-400 text-[11px] m-0" }, `🏢 ${u.company?.name || "Sin empresa"}`)
                            ),
                            React.createElement("div", { className: "mt-2 pt-2 border-t border-slate-700/50 flex justify-between items-center text-[10px] text-sky-300/80" },
                                React.createElement("span", null, `📍 ${u.address?.city || "Ciudad"}`),
                                React.createElement("span", null, `🌐 ${u.website}`)
                            )
                        )
                    )
                )
        )
    );
}

// Montaje en el contenedor DOM (#app)
const mountPoint = document.getElementById("app") || document.getElementById("sandbox-container");
if (mountPoint) {
    const root = ReactDOM.createRoot(mountPoint);
    root.render(React.createElement(CatalogoUsuariosAPI));
    console.log("⚛️ [React 18] CatalogoUsuariosAPI montado con éxito en el DOM.");
}
