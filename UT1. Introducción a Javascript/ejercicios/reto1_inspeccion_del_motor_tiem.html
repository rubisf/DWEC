Reto 1: Inspección del Motor, Tiempos de Ejecución y Prioridades del Event Loop

📝 Enunciado:

Diseña un script que: 1) Extraiga y clasifique mediante expresiones regulares el nombre y versión exacta del motor de JavaScript y navegador a partir de `navigator.userAgent`, 2) Evalúe la identidad del objeto global unificado (`window === globalThis`), y 3) Demuestre cuantitativamente con `performance.now()` la prioridad del Event Loop programando una macrotarea (`setTimeout 0ms`) antes de una microtarea (`Promise.resolve()` y `queueMicrotask()`), registrando una tabla de traza y validando el orden secuencial de resolución.

💡 Pistas: Usa expresiones regulares como /Chrome\/([0-9.]+)/, /Firefox\/([0-9.]+)/ o /Version\/([0-9.]+).*Safari/ sobre navigator.userAgent. Almacena en un array de traza cada evento con performance.now() y muestra console.table().

La especificación HTML/W3C define que el Event Loop procesa todo el Call Stack síncrono primero, luego vacía completamente la cola de microtareas (Promesas y queueMicrotask) y finalmente, en el siguiente ciclo, atiende la cola de macrotareas (como temporizadores de setTimeout).

function auditarMotorYEventLoop() {
    console.group("🔍 1. Detección y Extracción del Motor JS mediante navigator.userAgent");
    const ua = navigator.userAgent;
    let motorDetectado = "Desconocido";
    let versionMotor = "Desconocida";
    let navegador = "Navegador no identificado";

    // Expresiones regulares para extraer motor y versión exacta
    if (/Edg\/([0-9.]+)/i.test(ua)) {
        motorDetectado = "V8 (Blink)";
        versionMotor = ua.match(/Edg\/([0-9.]+)/i)[1];
        navegador = "Microsoft Edge";
    } else if (/Chrome\/([0-9.]+)/i.test(ua)) {
        motorDetectado = "V8 (Blink)";
        versionMotor = ua.match(/Chrome\/([0-9.]+)/i)[1];
        navegador = "Google Chrome / Chromium";
    } else if (/Firefox\/([0-9.]+)/i.test(ua)) {
        motorDetectado = "SpiderMonkey (Gecko)";
        versionMotor = ua.match(/Firefox\/([0-9.]+)/i)[1];
        navegador = "Mozilla Firefox";
    } else if (/Version\/([0-9.]+).*Safari/i.test(ua)) {
        motorDetectado = "JavaScriptCore (Nitro/WebKit)";
        versionMotor = ua.match(/Version\/([0-9.]+)/i)[1];
        navegador = "Apple Safari";
    }

    console.log("Navegador:", navegador);
    console.log("Motor de JavaScript:", motorDetectado);
    console.log("Versión detectada:", versionMotor);
    console.log("User-Agent en bruto:", ua);
    console.groupEnd();

    console.group("🌐 2. Comparativa de Identidad del Objeto Global");
    const auditoriaGlobal = {
        "window === globalThis": window === globalThis,
        "self === globalThis": typeof self !== "undefined" && self === globalThis,
        "typeof window": typeof window,
        "typeof globalThis": typeof globalThis,
        "Entorno estandarizado": "globalThis (ES2020)"
    };
    console.table(auditoriaGlobal);
    console.groupEnd();

    console.group("⚡ 3. Demostración Cuantitativa de Prioridades del Event Loop");
    const t0 = performance.now();
    const traza = [];

    function registrarPaso(fase, tipo, prioridadTeorica) {
        traza.push({
            "Orden de Ejecución": traza.length + 1,
            "Fase / Cola": fase,
            "Tipo de Tarea": tipo,
            "Prioridad": prioridadTeorica,
            "Timestamp Relativo (ms)": Number((performance.now() - t0).toFixed(4))
        });
    }

    // 1. Sentencia Síncrona Inicial en Call Stack
    registrarPaso("Call Stack", "Síncrono (Inicio)", "1 - Inmediata");

    // 2. Macrotarea: Se programa PRIMERO en el código deliberadamente
    setTimeout(() => {
        registrarPaso("Macrotask Queue", "Macrotarea (setTimeout 0ms)", "3 - Baja");

        // Al ejecutarse la macrotarea, el ciclo ha concluido y mostramos la tabla
        console.table(traza);
        const ordenCorrecto = traza[2]["Tipo de Tarea"].includes("Microtarea") && 
                              traza[traza.length - 1]["Tipo de Tarea"].includes("Macrotarea");
        console.log("¿Se demostró la prioridad de microtareas frente a macrotareas?:", ordenCorrecto ? "SÍ ✅" : "NO ❌");
        console.groupEnd();
    }, 0);

    // 3. Microtarea (Promise.resolve): Se programa DESPUÉS de setTimeout
    Promise.resolve().then(() => {
        registrarPaso("Microtask Queue", "Microtarea (Promise.resolve)", "2 - Alta");
    });

    // 4. Microtarea Nativa (queueMicrotask)
    if (typeof queueMicrotask === "function") {
        queueMicrotask(() => {
            registrarPaso("Microtask Queue", "Microtarea (queueMicrotask)", "2 - Alta");
        });
    }

    // 5. Sentencia Síncrona Final en Call Stack
    registrarPaso("Call Stack", "Síncrono (Fin)", "1 - Inmediata");
}

auditarMotorYEventLoop();
