Reto 5: Validador de DNI y Contraseñas Seguras con RegExp y Constraint Validation API

📝 Enunciado:

La Constraint Validation API en combinación con expresiones regulares y algoritmos de control matemático previene el envío de datos corruptos al backend sin romper la experiencia fluida del usuario.

// 1. Validador Algorítmico de DNI Español
function validarDNI(inputDNI) {
    const valor = inputDNI.value.trim().toUpperCase();
    const regexFormato = /^[0-9]{8}[A-Z]$/;

    if (!regexFormato.test(valor)) {
        inputDNI.setCustomValidity("El DNI debe componerse de 8 dígitos y una letra mayúscula final.");
        return false;
    }

    const numero = parseInt(valor.slice(0, 8), 10);
    const letraIngresada = valor.charAt(8);
    const tablaLetras = "TRWAGMYFPDXBNJZSQVHLCKE";
    const letraCalculada = tablaLetras[numero % 23];

    if (letraIngresada !== letraCalculada) {
        inputDNI.setCustomValidity(`Letra de control inválida. Para ${numero} corresponde la letra '${letraCalculada}'.`);
        return false;
    }

    inputDNI.setCustomValidity(""); // Campo 100% válido
    console.log("✅ DNI validado con éxito:", valor);
    return true;
}

// 2. Auditor Reactivo de Complejidad de Contraseña
function auditarSeguridadContrasena(inputPassword) {
    inputPassword.addEventListener("input", () => {
        const pass = inputPassword.value;
        const faltantes = [];

        if (pass.length < 8) faltantes.push("mínimo 8 caracteres");
        if (!/[A-Z]/.test(pass)) faltantes.push("al menos una mayúscula");
        if (!/[a-z]/.test(pass)) faltantes.push("al menos una minúscula");
        if (!/[0-9]/.test(pass)) faltantes.push("al menos un dígito numérico");
        if (!/[@$!%*?&._-]/.test(pass)) faltantes.push("un carácter especial (@$!%*?&._-)");

        if (faltantes.length > 0) {
            inputPassword.setCustomValidity("Requisitos pendientes: " + faltantes.join(", "));
            inputPassword.style.borderColor = "#ef4444";
        } else {
            inputPassword.setCustomValidity(""); // Conforme
            inputPassword.style.borderColor = "#10b981";
            console.log("✅ Contraseña robusta verificada.");
        }
    });
}
