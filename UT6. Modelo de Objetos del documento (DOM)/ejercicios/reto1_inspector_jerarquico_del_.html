Comprender la anatomía del árbol de nodos y distinguir entre nodos de texto y nodos de elemento es indispensable para manipular el DOM sin errores derivados de saltos de línea.

function inspeccionarJerarquiaDOM(nodo, nivel = 0) {
    const indentacion = "  ".repeat(nivel);
    const tipoTexto = nodo.nodeType === Node.ELEMENT_NODE ? "Etiqueta" :
                      nodo.nodeType === Node.TEXT_NODE ? "Texto" :
                      nodo.nodeType === Node.COMMENT_NODE ? "Comentario" : "Otro";

    // Si es un nodo de texto con sólo espacios en blanco, omitir o resumir
    if (nodo.nodeType === Node.TEXT_NODE) {
        const textoLimpio = nodo.nodeValue.trim();
        if (textoLimpio) {
            console.log(\`\${indentacion}📄 [Texto]: "\${textoLimpio}"\`);
        }
        return;
    }

    if (nodo.nodeType === Node.ELEMENT_NODE) {
        console.log(\`\${indentacion}📁 <\${nodo.tagName.toLowerCase()}> [ID: \${nodo.id || "sin id"}, Hijos: \${nodo.children.length}]\`);
        
        // Recorrido recursivo de los hijos de elemento
        for (const hijo of nodo.childNodes) {
            inspeccionarJerarquiaDOM(hijo, nivel + 1);
        }
    }
}
