Encapsular modales asíncronos en Promesas permite mantener una sintaxis secuencial limpia con async/await sin sufrir el bloqueo del hilo de ejecución característico de alert o confirm.

// Gestor asíncrono de confirmaciones basado en Promesas y <dialog>
function confirmarAccionModal(mensaje) {
    return new Promise((resolve) => {
        // Crear elemento dialog dinámico si no existe
        let dialog = document.getElementById('modal-confirmacion-dinamico');
        if (!dialog) {
            dialog = document.createElement('dialog');
            dialog.id = 'modal-confirmacion-dinamico';
            dialog.style.cssText = "border: none; border-radius: 12px; padding: 20px; box-shadow: 0 10px 30px rgba(0,0,0,0.3); font-family: sans-serif;";
            document.body.appendChild(dialog);
        }

        dialog.innerHTML = \`
            <div style="min-width: 300px;">
                <h3 style="margin-top: 0; color: #0f172a;">Confirmación Requerida</h3>
                <p style="color: #475569; font-size: 0.9em;">\${mensaje}</p>
                <div style="display: flex; justify-content: flex-end; gap: 10px; margin-top: 20px;">
                    <button id="btn-cancelar" style="padding: 8px 14px; border: 1px solid #cbd5e1; background: #f8fafc; border-radius: 6px; cursor: pointer;">Cancelar</button>
                    <button id="btn-aceptar" style="padding: 8px 14px; border: none; background: #2563eb; color: white; border-radius: 6px; cursor: pointer;">Aceptar</button>
                </div>
            </div>
        \`;

        dialog.querySelector('#btn-cancelar').onclick = () => {
            dialog.close();
            resolve(false);
        };

        dialog.querySelector('#btn-aceptar').onclick = () => {
            dialog.close();
            resolve(true);
        };

        dialog.showModal();
    });
}

// Ejemplo de consumo con async/await
async function ejecutarEliminacion() {
    const respuesta = await confirmarAccionModal("¿Deseas purgar la base de datos temporal?");
    console.log("Resultado de confirmación asíncrona:", respuesta);
}
