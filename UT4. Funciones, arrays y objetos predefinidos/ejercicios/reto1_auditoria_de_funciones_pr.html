Reto 1: Auditoría de Funciones Predefinidas, Ámbito Léxico de this y Procesador de URLs

📝 Enunciado:

Diseña un script que: 1) Audite y demuestre el comportamiento de las funciones predefinidas globales (`parseInt` con base explícita, `parseFloat`, la diferencia entre `isNaN` y `Number.isNaN`, `isFinite`, y la codificación/decodificación segura con `encodeURIComponent` y `decodeURIComponent`), 2) Contraste empíricamente el enlace léxico de `this` en una Función Flecha (Arrow Function) frente al enlace dinámico en una Declaración de Función tradicional, y 3) Construya una función `construirURLConsulta(baseURL, endpoint, ...parametros)` que reciba parámetros rest, sanitice y codifique los pares clave-valor y ensamble la URL final.

💡 Pistas: Usa Number.isNaN("abc") (false) vs isNaN("abc") (true) para contrastar la coerción previa. Diseña un objeto con un método clásico y una función flecha interna para demostrar cómo la flecha retiene el this léxico.

Las funciones globales predefinidas gestionan transformaciones elementales seguras, mientras que las funciones flecha resuelven el problema histórico de la pérdida de contexto de this en callbacks asíncronos.

function laboratorioFunciones() {
    console.group("🔍 1. Auditoría de Funciones Predefinidas Globales");
    
    // parseInt con base explícita vs parseFloat
    const enteroParseado = parseInt("100px", 10);
    const floatParseado = parseFloat("3.14159265rad");
    console.log("parseInt('100px', 10):", enteroParseado);
    console.log("parseFloat('3.14159265rad'):", floatParseado);

    // Comparativa crítica: isNaN vs Number.isNaN
    console.log("isNaN('texto'):", isNaN("texto"), "(coacciona a NaN primero)");
    console.log("Number.isNaN('texto'):", Number.isNaN("texto"), "(estricto: no coacciona)");
    console.log("Number.isFinite(100 / 0):", Number.isFinite(100 / 0), "(Infinity no es finito)");

    // Codificación y decodificación de URLs
    const querySensible = "categoría=desarrollo web & cliente=100%";
    const codificado = encodeURIComponent(querySensible);
    const decodificado = decodeURIComponent(codificado);
    console.log("encodeURIComponent:", codificado);
    console.log("decodeURIComponent:", decodificado);
    console.groupEnd();

    console.group("🎯 2. Enlace Léxico de 'this': Función Clásica vs Función Flecha");
    const componenteUsuario = {
        nombre: "Docente DWEC",
        rol: "Profesor",
        
        // Método con Function Declaration clásica: su propio 'this' depende de quién lo llame
        metodoClasico: function() {
            return `Usuario clásico: ${this.nombre} (${this.rol})`;
        },

        // Demostración de flecha reteniendo el 'this' léxico envolvente
        ejecutarConCallback: function() {
            // En una función clásica interna, 'this' se perdería a undefined o window en modo estricto.
            // La función flecha captura estáticamente el 'this' del método envolvente:
            const callbackFlecha = () => {
                return `Captura léxica exitosa: ${this.nombre} - ${this.rol}`;
            };
            return callbackFlecha();
        }
    };

    console.log(componenteUsuario.metodoClasico());
    console.log(componenteUsuario.ejecutarConCallback());
    console.groupEnd();

    console.group("🔗 3. Procesador de URLs con Parámetros Rest y Sanitización");
    const construirURLConsulta = (baseURL, endpoint, ...parametros) => {
        const urlLimpia = baseURL.replace(/\/$/, "") + "/" + endpoint.replace(/^\//, "");
        if (parametros.length === 0) return urlLimpia;

        const pares = parametros.map(([clave, valor]) => {
            const k = encodeURIComponent(String(clave).trim());
            const v = encodeURIComponent(String(valor).trim());
            return `${k}=${v}`;
        });

        return `${urlLimpia}?${pares.join("&")}`;
    };

    const urlGenerada = construirURLConsulta(
        "https://api.formacion.es/v1",
        "cursos/buscar",
        ["modulo", "Desarrollo Web en Entorno Cliente"],
        ["notaMinima", 7],
        ["orden", "prioritario & seguro"]
    );
    console.log("URL Final Generada y Sanitizada:");
    console.log(urlGenerada);
    console.groupEnd();
}

laboratorioFunciones();
